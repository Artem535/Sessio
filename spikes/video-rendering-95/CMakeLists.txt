cmake_minimum_required(VERSION 3.28)
project(SessioVideoRenderingBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets)
add_executable(sessio-video-rendering-benchmark main.cpp)
target_link_libraries(sessio-video-rendering-benchmark PRIVATE Qt6::Widgets Qt6::OpenGLWidgets)

# Optional offline SDK probe, using an already downloaded SDK. No downloads.
if(DEFINED LiveKit_DIR)
  find_package(LiveKit CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(sessio-video-stream-probe sdk_stream_probe.cpp)
  target_link_libraries(sessio-video-stream-probe PRIVATE LiveKit::livekit Threads::Threads)
  set(LIVEKIT_CURL_LIBRARY "" CACHE FILEPATH "libcurl matching the prebuilt SDK ABI")
  if(NOT LIVEKIT_CURL_LIBRARY)
    message(FATAL_ERROR "Set LIVEKIT_CURL_LIBRARY to the SDK-compatible libcurl; see Sessio src/video/CMakeLists.txt")
  endif()
  target_link_libraries(sessio-video-stream-probe PRIVATE "${LIVEKIT_CURL_LIBRARY}")
endif()
